class NewJ4Decoder(unittest.TestCase):
    def test_exact_original_bytes_and_published_time_preserved(self):
        f = Fixture();out = f.decode()
        self.assertEqual(out.writer_raw, c.canonical(f.writer) + b'\n')
        self.assertEqual(out.linked_receipt.raw, out.writer_raw)
        self.assertEqual(out.core_receipt.raw, f.stdout())
        self.assertEqual(out.linked_receipt.completed_at, d.instant(f.writer['published_at']))
        self.assertEqual(out.linked_receipt.request_raw, f.request)
        self.assertEqual(f.calls, ['approval', 'original'])
        self.assertFalse(f.process['operational_GO'])
    def test_explicit_j_source_selector_is_conserved(self):
        f = Fixture();f.rule['linked_source_sha256'] = d.SOURCE_PINS['j_slot']
        self.assertEqual(f.decode().linked_receipt.source_sha256, d.SOURCE_PINS['j_slot'])
    def test_unknown_source_selector_refused(self):
        f=Fixture();f.rule['linked_source_sha256']=hash_name('unknown')
        with self.assertRaisesRegex(c.Hold, 'J4_APPROVAL_ABI'):f.adapter()
    def test_source_byte_mutation_refused_before_callbacks(self):
        f=Fixture();f.sources=((f.sources[0][0],f.sources[0][1]+b'\n'),)+f.sources[1:]
        with self.assertRaisesRegex(c.Hold,'J4_APPROVED_SOURCE_CHANGED'):f.adapter()
        self.assertEqual(f.calls,[])
    def test_missing_independent_original_verifier_refused(self):
        f=Fixture();raw=c.canonical(f.rule)
        with self.assertRaisesRegex(c.Hold,'J4_INDEPENDENT_VERIFIERS_REQUIRED'):
            d.J4ReceiptAdapter(d.Binding(f.registry_raw,f.request,f.bound,f.sources),approval=d.Approval(raw,c.sha(raw),f.derivation),verify_approval=None,verify_original=None)
    def test_real_cannot_promote_fixture_verifier(self):
        f=Fixture();f.rule['mode']='REAL'
        with self.assertRaisesRegex(c.Hold,'EXTERNAL_VERIFIER_UNBOUND'):f.adapter()
    def test_boolean_is_not_original_attestation(self):
        f=Fixture();f.original_callback=lambda *args:True
        with self.assertRaisesRegex(c.Hold,'EXTERNAL_VERIFIER_BOOLEAN_IS_NOT_ATTESTATION'):f.decode()
    def test_independent_origin_rejection_refuses_self_consistent_fields(self):
        f=Fixture()
        def reject(*args):raise c.Hold('FIXTURE_ORIGIN_NOT_PROVEN')
        f.original_callback=reject
        with self.assertRaisesRegex(c.Hold,'FIXTURE_ORIGIN_NOT_PROVEN'):f.decode()
    def test_foreign_invocation_bound_refused(self):
        f=Fixture();f.invocation=c.Invocation('admission_manifest',c.context(f.q),c.sha(f.request),hash_name('foreign'),f.now+timedelta(seconds=30),time.monotonic()+30,())
        with self.assertRaisesRegex(c.Hold,'J4_INVOCATION_UNBOUND'):f.decode()
    def test_request_registry_selector_change_refused(self):
        f=Fixture();f.registry['originals']['request']['sha256']=hash_name('foreign');f.registry_raw=c.canonical(f.registry)
        with self.assertRaisesRegex(c.Hold,'J4_REGISTRY_PIN_CHANGED'):f.adapter()
    def test_family_key_cannot_be_outer_operation_key(self):
        f=Fixture();f.slot['attempt_key']=f.process['outer_attempt_key']
        with self.assertRaisesRegex(c.Hold,'J4_SLOT_NOT_COMPLETE'):f.decode()
    def test_uncertain_terminal_never_yields_linked_complete(self):
        f=Fixture();f.slot['status']='UNCERTAIN'

    def test_already_published_is_not_new_j_complete(self):
