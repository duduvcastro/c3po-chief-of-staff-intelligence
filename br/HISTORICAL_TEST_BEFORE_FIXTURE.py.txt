class BoundedTests(unittest.TestCase):
    def mutated(self,f,mutation):
        value=json.loads(f.raw);mutation(value);return a.canonical(value)
    def reject(self,call,code=None):
        with self.assertRaises((a.Hold,core.Hold)) as error:call()
        if code:self.assertEqual(str(error.exception),code)

    def test_large_twenty_megabyte_journal_becomes_limited_readback(self):
        f=Fixture(padding_records=420)
        self.assertGreater(len(a.canonical(f.records)),20*1024*1024)
        self.assertLess(len(f.raw),12_000)
        decoded=f.decode()
        result=a.check_capacity_manifest(f.scope,f.receipt,decoded.row,decoded.journal,f.manifest)
        self.assertEqual(result['state_sha256'],f.row['state_sha'])
        self.assertNotEqual(a.digest(decoded.row['state']),f.row['state_sha'])
        self.assertNotIn('ledger',decoded.row['state'])
        self.assertEqual(decoded.journal.count,421)

    def test_native_integral_source_definitions_exact(self):
        self.assertEqual(a.sha((Path(b.__file__).parent/'sources/r2d2_v2_store.py').read_bytes()),b.STORE_SHA)
        self.assertIn('verify_journal',b._native_verifier())

    def test_after_capture_keeps_original_universe_and_copy(self):

    def test_bounded_bridge_checks_real_phase_and_native_clocks(self):

        f.scope=replace(f.scope,finite_authority_sha256=down.q['authority_sha256'])
        f.authority=down.authority

        f.request=a.sha(core.canonical(down.q));f.bound=a.sha(down.bundle.bound)
        f.raw=f.expected=f.produce()
        directory='/fixture-private-journal'
        # Exact actual epoch catalog schema, independent of the DB projection.
        catalog=a.FileEvidence(a.canonical({'schema':'MASSIVE_SESSION_ROOT_V1','epoch':f.scope.epoch,
                     'device':1,'inode':44}),1,9,0,0,0o600,1)
        snapshot=g.BoundedSnapshot(a.canonical(f.receipt),f.raw,f.manifest,f.at,catalog,f.root,directory)
        gate=g.BoundedImageCapacityGate(f.scope,lambda *_:snapshot,lambda *_:None,journal_directory=directory,

            self.reject(lambda:gate(down.q,f.phase,f.at,task),'IMAGE_SNAPSHOT_NOT_FRESH')
